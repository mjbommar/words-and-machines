%% ============================================================
%% figure-preamble.tex — shared preamble for standalone TikZ sources
%% under latex/figures/src/*.tikz (compiled by scripts/build_tikz.py).
%%
%% Each source is wrapped in a standalone document that inputs this file,
%% so every diagram renders with the paper's palette and serif and
%% responds to \GrayscaleMode exactly as the body does. Compiles with
%% pdflatex (Type 1 fonts) for portability; scripts/build_tikz.py may
%% pass \def\GrayscaleMode{} for a B&W variant.
%% ============================================================

\usepackage{xcolor}
\usepackage{tikz}
\usepackage{pgfplots}
\pgfplotsset{compat=1.18}
\usetikzlibrary{arrows.meta, positioning, calc, fit, backgrounds,
  shapes.geometric, shapes.misc, decorations.pathreplacing}

% The paper's palette (primitives + semantic tokens), so figure colors
% match the callouts and collapse under \GrayscaleMode with them.
\input{\FigurePreambleDir preamble/colors.tex}

% Convenience aliases for common diagram color names, mapped onto the
% paper palette so a figure stays in-house and grayscale-safe.
\colorlet{fig-ink}{text-primary}
\colorlet{fig-muted}{text-muted}
\colorlet{fig-accent}{accent-500}
\colorlet{fig-accent-dark}{accent-700}
\colorlet{fig-fill}{accent-50}
\colorlet{fig-highlight}{highlight}
