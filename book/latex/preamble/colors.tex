%% ============================================================
%% preamble/colors.tex — 4-layer color system (ADR 0005)
%%
%% Purpose      : defines every color the book may use, in layers:
%%                  1. PRIMITIVES  gray-50..950 + accent-50..950 scales
%%                  2. SEMANTICS   purpose-named tokens (text-primary, …)
%%                  3. COMPONENTS  per-element bindings (bk-*, code-*)
%%                (Layer 4, legacy aliases, is deliberately absent —
%%                the template starts clean.)
%% Dependencies : xcolor. Nothing chapter-facing: chapters never name
%%                colors (authoring contract, "hard-coded colors"
%%                forbidden). Safe under cover.tex's article preamble.
%% Provenance   : architecture matured in vibe-coding-for-lawyers,
%%                ai-law-finance and the beamer template; the "names
%%                must stay truthful" rule reacts to the RFC book's
%%                cyan-* tokens ending up holding gold (ADR 0005).
%%
%% RETHEMING = edit the primitive hex values below. Semantic and
%% component layers reference names only and never need touching.
%% \GrayscaleMode (B&W POD ink) collapses the accent scale onto the
%% gray scale AT THE PRIMITIVE LAYER, so every downstream token is
%% automatically press-safe — no per-component overrides.
%% ============================================================

% Cover identity (consumed by latex/cover/cover.tex; must keep
% title/subtitle/body legible on cover-field in BOTH color and
% grayscale builds — field 950 vs text 50/100/200 holds either way).
%
% cover-field is defined in CMYK, not from the accent ramp: it fills
% the whole wrap, and POD presses cap total ink (C+M+Y+K) at ~240%
% (`make cover-ink` gates this). RGB accent-950 converts to ~274%.
% If you swap the accent palette, re-derive an equivalent dark CMYK
% with channel sum <= 2.30 and re-run the gate.
\definecolor{cover-field}{cmyk}{0.72,0.75,0.28,0.55}  % deep indigo, TAC 230%
\colorlet{cover-accent}{accent-500}   % accent band / spine stripe
\colorlet{cover-title}{gray-50}       % title text on field
\colorlet{cover-subtitle}{accent-200} % secondary text on field
\colorlet{cover-body}{gray-100}       % back-cover body text on field
